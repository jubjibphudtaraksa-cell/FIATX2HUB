local Tabs = {}
local CurrentTab
local Minimized = false

local function Highlight(text)
	text = text:gsub("&","&amp;")
	text = text:gsub("<","&lt;")
	text = text:gsub(">","&gt;")
	text = text:gsub("(%-%-.-)\n","<font color='#5c6370'>%1</font>\n")
	text = text:gsub("(%f[%a])(local|function|end|if|then|else|for|while|do|return|repeat|until|in)(%f[%A])","<font color='#c678dd'>%2</font>")
	text = text:gsub('(".-")',"<font color='#98c379'>%1</font>")
	text = text:gsub("(%d+)","<font color='#e5c07b'>%1</font>")
	return text
end

local function SelectTab(tab)
	for _,v in pairs(Tabs) do
		v.Box.Visible = false
		v.Button.TextColor3 = Color3.fromRGB(180,180,180)
	end
	CurrentTab = tab
	tab.Box.Visible = true
	tab.Button.TextColor3 = Color3.fromRGB(0,255,140)
end

local function CreateTab()
	local index = #Tabs + 1

	local Btn = Instance.new("TextButton", TabBar)
	Btn.Size = UDim2.new(0,80,1,0)
	Btn.Position = UDim2.new(0,(index-1)*85+10,0,0)
	Btn.Text = "Tab "..index
	Btn.Font = Enum.Font.Gotham
	Btn.TextSize = 12
	Btn.TextColor3 = Color3.fromRGB(180,180,180)
	Btn.BackgroundTransparency = 1

	local Box = Instance.new("TextBox", Content)
	Box.Size = UDim2.new(1,-10,1,-10)
	Box.Position = UDim2.new(0,5,0,5)
	Box.BackgroundTransparency = 1
	Box.MultiLine = true
	Box.ClearTextOnFocus = false
	Box.RichText = true
	Box.TextWrapped = false
	Box.TextXAlignment = Enum.TextXAlignment.Left
	Box.TextYAlignment = Enum.TextYAlignment.Top
	Box.Font = Enum.Font.Code
	Box.TextSize = 15
	Box.TextColor3 = Color3.fromRGB(230,230,230)
	Box.Text = "-- Tab "..index
	Box.Visible = false

	local lock = false
	Box:GetPropertyChangedSignal("Text"):Connect(function()
		if lock then return end
		lock = true
		local raw = Box.Text:gsub("<.->","")
		local pos = Box.CursorPosition
		Box.Text = Highlight(raw)
		Box.CursorPosition = pos
		lock = false
	end)

	local Data = {Button=Btn,Box=Box}
	table.insert(Tabs,Data)

	Btn.MouseButton1Click:Connect(function()
		SelectTab(Data)
	end)

	if #Tabs == 1 then
		SelectTab(Data)
	end
end

CreateTab()

AddTab.MouseButton1Click:Connect(CreateTab)

	if not CurrentTab then return end
	local code = CurrentTab.Box.Text:gsub("<.->","")
	pcall(function()

	end)
end)

Clear.MouseButton1Click:Connect(function()
	if CurrentTab then
		CurrentTab.Box.Text = ""
	end
end)

Save.MouseButton1Click:Connect(function()
	if CurrentTab then
		setclipboard(CurrentTab.Box.Text:gsub("<.->",""))
	end
end)
